
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(100) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--------------------------------------------------------

CREATE OR REPLACE PROCEDURE sp_insert_user(
    p_username VARCHAR,
    p_email VARCHAR,
    p_password VARCHAR
)
LANGUAGE plpgsql
AS $$
BEGIN
    INSERT INTO users (username, email, password)
    VALUES (p_username, p_email, p_password);
END;
$$;


-------------------------------------------------------------------


CREATE OR REPLACE PROCEDURE sp_update_user(
    p_id INT,
    p_username VARCHAR,
    p_email VARCHAR,
    p_password VARCHAR
)
LANGUAGE plpgsql
AS $$
BEGIN
    UPDATE users
    SET username = p_username,
        email = p_email,
        password = p_password
    WHERE id = p_id;
    
    IF NOT FOUND THEN
        RAISE EXCEPTION 'Utilisateur avec l''ID % introuvable', p_id;
    END IF;
END;
$$;


-----------------------------------------------------------------------------------


CREATE OR REPLACE PROCEDURE sp_delete_user(
    p_id INT
)
LANGUAGE plpgsql
AS $$
BEGIN
    DELETE FROM users
    WHERE id = p_id;
    
    IF NOT FOUND THEN
        RAISE EXCEPTION 'Utilisateur avec l''ID % introuvable', p_id;
    END IF;
END;
$$;
